* Overview

A hands-on learning project to master SIMD (Single Instruction Multiple Data), inline assembly, and GPU compute through building high-performance image and audio filters.

** Goals
- Learn ~core::arch~ SIMD intrinsics (stable Rust)
- Write inline assembly and understand x86-64 ISA
- Master GPU compute shaders with wgpu
- Achieve 10-100x speedups through optimization
- Build reusable knowledge for audio DSP work

** Why Image/Audio Filtering?
- Clear correctness (compare output to reference implementation)
- Obvious performance gains (measurable speedup)
- Works well for SIMD, assembly, AND GPU
- Practical applications (can apply to Tunes later)
- Visual/audible feedback (satisfying to see results)

#+BEGIN_SRC rust
fn box_blur_3x3(image: &[f32], width: usize, height: usize) -> Vec<f32> {
    let mut output = vec![0.0; image.len()];

    for y in 1..height-1 {
        for x in 1..width-1 {
            let idx = y * width + x;

            // Average 3x3 neighborhood
            let sum =
                image[idx - width - 1] + image[idx - width] + image[idx - width + 1] +
                image[idx - 1]         + image[idx]         + image[idx + 1] +
                image[idx + width - 1] + image[idx + width] + image[idx + width + 1];

            output[idx] = sum / 9.0;
        }
    }

    output
}
#+END_SRC

*** Example: SIMD Setup
#+BEGIN_SRC rust
#[cfg(target_arch = "x86_64")]
fn blur_simd(image: &[f32], width: usize, height: usize) -> Vec<f32> {
    unsafe {
        use core::arch::x86_64::*;

        let mut output = vec![0.0; image.len()];

        // Process 8 pixels at once with AVX2
        for y in 1..height-1 {
            let mut x = 1;

            // Main loop: process 8 pixels at a time
            while x + 8 < width - 1 {
                let idx = y * width + x;

                // Load 8 pixels from each position
                let top_left = _mm256_loadu_ps(&image[idx - width - 1]);
                let top = _mm256_loadu_ps(&image[idx - width]);
                // ... load all 9 positions

                // Add them all
                let sum = _mm256_add_ps(top_left, top);
                // ... add remaining

                // Divide by 9
                let nine = _mm256_set1_ps(9.0);
                let result = _mm256_div_ps(sum, nine);

                // Store result
                _mm256_storeu_ps(&mut output[idx], result);

                x += 8;
            }

            // Handle remaining pixels with scalar code
            // ...
        }

        output
    }
}
#+END_SRC

*** Key Intrinsics to Learn
- ~_mm256_loadu_ps~ - Load unaligned floats
- ~_mm256_storeu_ps~ - Store unaligned floats
- ~_mm256_add_ps~ - Add 8 floats in parallel
- ~_mm256_mul_ps~ - Multiply 8 floats in parallel
- ~_mm256_div_ps~ - Divide 8 floats in parallel
- ~_mm256_set1_ps~ - Broadcast scalar to vector

*** Example: Inline Assembly Template
#+BEGIN_SRC rust
fn blur_asm(image: &[f32], width: usize, height: usize) -> Vec<f32> {
    let mut output = vec![0.0; image.len()];

    #[cfg(target_arch = "x86_64")]
    unsafe {
        use core::arch::asm;

        let src = image.as_ptr();
        let dst = output.as_mut_ptr();

        asm!(
            // Load 8 floats into ymm0
            "vmovups ymm0, [{src}]",

            // Load 8 floats into ymm1
            "vmovups ymm1, [{src} + 32]",

            // Add them
            "vaddps ymm0, ymm0, ymm1",

            // Store result
            "vmovups [{dst}], ymm0",

            src = in(reg) src,
            dst = in(reg) dst,
            out("ymm0") _,
            out("ymm1") _,
            options(nostack),
        );
    }

    output
}
#+END_SRC

*** Example: WGSL Compute Shader
#+BEGIN_SRC wgsl
@group(0) @binding(0) var<storage, read> input: array<f32>;
@group(0) @binding(1) var<storage, read_write> output: array<f32>;
@group(0) @binding(2) var<uniform> params: Params;

struct Params {
    width: u32,
    height: u32,
}

@compute @workgroup_size(16, 16)
fn blur_kernel(@builtin(global_invocation_id) id: vec3<u32>) {
    let x = id.x;
    let y = id.y;

    if (x >= params.width || y >= params.height) {
        return;
    }

    // Skip edges
    if (x == 0u || y == 0u || x >= params.width - 1u || y >= params.height - 1u) {
        let idx = y * params.width + x;
        output[idx] = input[idx];
        return;
    }

    let idx = y * params.width + x;

    // 3x3 box blur
    var sum = 0.0;
    for (var dy = -1; dy <= 1; dy++) {
        for (var dx = -1; dx <= 1; dx++) {
            let sample_idx = (y + u32(dy)) * params.width + (x + u32(dx));
            sum += input[sample_idx];
        }
    }

    output[idx] = sum / 9.0;
}
#+END_SRC

//! Scalar baseline implementations - intentionally simple and slow

/// 3x3 box blur - averages each pixel with its 8 neighbors
pub fn box_blur_3x3(image: &[f32], width: usize, height: usize) -> Vec<f32> {
    let mut output = vec![0.0; image.len()];

    // Skip edges (simplification for now)
    for y in 1..height-1 {
        for x in 1..width-1 {
            let idx = y * width + x;

            let sum =
                image[idx - width - 1] + image[idx - width] + image[idx - width + 1] +
                image[idx - 1]         + image[idx]         + image[idx + 1] +
                image[idx + width - 1] + image[idx + width] + image[idx + width + 1];

            output[idx] = sum / 9.0;
        }
    }

    output
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn test_box_blur_simple() {
        // 5x5 image, all ones
        let image = vec![1.0; 25];
        let blurred = box_blur_3x3(&image, 5, 5);

        // Center pixels should be 1.0 (average of 1.0s)

** First Benchmark (benches/blur.rs)

#+BEGIN_SRC rust
use criterion::{black_box, criterion_group, criterion_main, Criterion, BenchmarkId};
use fast_filters::scalar;

fn benchmark_blur(c: &mut Criterion) {
    let mut group = c.benchmark_group("box_blur_3x3");

    for size in [256, 512, 1024, 2048].iter() {
        let image = vec![1.0f32; size * size];

        group.bench_with_input(
            BenchmarkId::new("scalar", size),
            size,
            |b, &size| {
                b.iter(|| {
                    scalar::box_blur_3x3(
                        black_box(&image),
                        black_box(size),
                        black_box(size),
                    )
                });
            },
        );
    }

    group.finish();
}

criterion_group!(benches, benchmark_blur);
criterion_main!(benches);
#+END_SRC

* Notes

** Why f32 instead of u8?
- Easier to work with in SIMD (no overflow concerns)
- Can convert to/from u8 for image I/O
- More representative of audio processing

** Why Box Blur First?
- Simplest possible filter
- All coefficients are equal (1/9)
- Easy to verify correctness
- Good learning foundation

** Performance Tips
- Always benchmark with ~--release~
- Use ~black_box~ to prevent optimization
- Test with realistic data sizes
- Profile before optimizing further
